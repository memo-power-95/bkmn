Function main
''''''''''''''''''''''''''''''' TOP_TLA_Station_07 '''''''''''''''''''''''''''''''''''''

'--- Load the robot point file ---  
LoadPoints "robot1.pts"
  
'--- initialization ---
Call Init
  
'--- Robot parameter settings ---
Tool 0
I1 = 100
Weight 5 'load setting ¸ºÔØÉè¶¨
Motor On 'Power up the motor µç»úÉÏµç
FineDist 0.1 'Target point positioning accuracy setting (mm) 
Power High 'High power mode setting  
Speed 100 'Internal speed setting (%)
SpeedS 2000 'Linear speed setting (mm/sec) 
SpeedFactor I1 'External speed setting (%) 
Accel 100, 100 'Acceleration and deceleration ratio setting (%) 
AccelS 15000, 8000 'Acceleration and deceleration value setting (mm/sec2) 
PTPBoost 75, 75, 75 'Nudge parameter setting


'--- Detect the status and start of background task No. 65 ----
If TaskState(65) = 0 Then
	Xqt 65, bgmain
EndIf
  
'--- Normal workflow (running in a loop)  ---
Do
  SpeedFactor I1 'External speed setting (%) 
  Select S2$
    Case "001"
      Call PhotoFeedTray(2) 'Taking pictures of the feeding tray 
    Case "002"
  	  Call ScanFeedTray(R1) 'Feed tray scan code 
  	Case "003"
  	  Call PhotoFeedDUT(5, R1) 'Taking pictures of incoming 
  	Case "004"
  	  Call ScanFeedDUT(69, R1)  'Incoming product scan 
    Case "005"
      Call PhotoDischargeTray(8) 'Taking pictures of the discharge 
    Case "006"
      Call ScanDischargeTray(R1) 'Outlet tray scan code
    Case "007"
      Call PhotoDischargeDUT(11, R1) 'Outlet tray product photo
    Case "008"
      Call ScanDischargeDUT(12, R1)  'Outlet tray product scan code
    Case "009"
      Call ScanDUTBelow(13, R2)      'Scan the code under the product
    Case "010"
      If B17 = True Then
          If R3 = 0 And R4 = 0 And R5 = 0 Then
      	    Call TakeFeedDUT(17, R1, R2, R3, R4, R5) 'Feed and take product(without vision)
      	  Else
      		Call TakeFeedDUT_1(17, R1, R2, R3, R4, R5)  'Feed and take product(with vision)
      	  EndIf
      	Else
      	Call TakeFeedDUT(17, R1, R2, R3, R4, R5) 'Feed and take product(without vision)
      EndIf
  	Case "011"
  	  If B17 = True Then
  	     If R3 = 0 And R4 = 0 And R5 = 0 Then
  		    Call PutDischargeDUT(21, R1, R2, R3, R4, R5) 'Take out the product(without vision)
  		 Else
  		    Call PutDischargeDUT_1(21, R1, R2, R3, R4, R5) 'Take out the product(with vision)
  		 EndIf
 	  Else
  	     Call PutDischargeDUT(21, R1, R2, R3, R4, R5) 'Take out the product(without vision)
  	  EndIf
    Case "013"
      Call BackSafe   'back to position
    Case "014"
      Call VisionCalibration(R1, R3, R4, R5)   'visual calibration
    Case "016"
      Call Throw_NG(R1, R2)   'NG position throwing
   Case "017"
      Call Photo_NG(R1)
    Case "020"
       Call Buffer_Take(R1, R2) 'Take product on the buffer position
    Case "021"
       Call Buffer_Put(R1, R2) 'Put product on the buffer position  

    Default
    Wait 0.01
  Send
Loop
    
Fend

Function Throw_NG(bli31 As Integer, bli32 As Integer)
 
  Off G1Down; Off G2Down; Off G3Down; On G1Up; On G2Up; On G3Up 'Gripper 1/2/3 up 
  Wait Sw(G1UpSignal) = On; Wait Sw(G2UpSignal) = On; Wait Sw(G3UpSignal) = On 'Wait for the gripper 1/2/3 to rise into place

'--- Check the gripper index ---
  If bli32 < 1 Or bli32 > 3 Then
	    S11$ = S2$ + "_E1620_a"; B1 = True
	    Print "Gripper index exception"
	    Pause
S2$ = ""
Exit Function
  EndIf

  '--- Check the gripper index ---

  If bli31 < 1 Or bli31 > 4 Then
	    S11$ = S2$ + "_E1620_a"; B1 = True
	    Print "Gripper index exception"
	    Pause
S2$ = ""
Exit Function
  EndIf
  
    Integer bli55
   If bli32 = 1 Then
     bli55 = 54
   ElseIf bli32 = 2 Then
     bli55 = 58
   ElseIf bli32 = 3 Then
     bli55 = 62
   EndIf
   bli55 = bli55 + bli31 - 1

  
	Hand P(bli55), Lefty
	Go P(bli55) /L
	Call GripperDown(R2)
	Call GripperOpen(R2)
	Wait 0.1
	Call GripperUp(R2)

S2$ = ""
S11$ = "016_OKOK_a"; B1 = True
Print "Robot NG throwing completed"
	
Fend

Function Message_analyze
	
	   '---------parsing strings---------    
Lable_33:
   S3$(0) = Left$(S10$, 3)
   S3$(1) = Mid$(S10$, 5, 3)
   S3$(2) = Mid$(S10$, 9, 2)
   S3$(3) = Mid$(S10$, 12, 7)
   S3$(4) = Mid$(S10$, 20, 7)

   S3$(6) = Right$(S10$, 1)
   
   S3$(7) = Mid$(S10$, 4, 1)
   S3$(8) = Mid$(S10$, 8, 1)
   S3$(9) = Mid$(S10$, 11, 1)
   S3$(10) = Mid$(S10$, 19, 1)

   S3$(12) = Mid$(S10$, 35, 1)
   
    If S3$(0) = "101" Then
    	Call WriteRobotDO
      GoTo Lable_3
   EndIf
   
       If S3$(0) = "102" Then
    		Call ReadRobotDO
      GoTo Lable_3
   EndIf
   
       If S3$(0) = "103" Then
    	Call ReadRobotDI
      GoTo Lable_3
   EndIf
   
   
   If S3$(0) = "104" Then
      R21 = Val(S3$(3))
      Call WriteRobotSpeed
   Else
	  R1 = Val(S3$(1))
	  R2 = Val(S3$(2))
	  R3 = Val(S3$(3))
	  R4 = Val(S3$(4))
	  R5 = Val(S3$(5))
	  S2$ = S3$(0)
   EndIf
   Wait 0.1
  
   
	'---------check function code--------
   I5 = Val(S3$(0))
   If I5 <= 0 Then
   Print "function code is " + S3$(0)
   S11$ = S2$ + "_E1619_a"; B1 = True
   B1 = True
   B19 = True
   Wait 0.1
   GoTo Lable_3
   EndIf

   S11$ = S2$ + "_E1618_a"; B1 = True
   B1 = True
   B19 = True
   Wait 0.1
   GoTo Lable_3
   EndIf
   
   '---------check the end string-----   	
   S4$ = Mid$(S10$, 36, 1)

   Wait 0.1
   GoTo Lable_3
      EndIf
